cmake_minimum_required(VERSION 3.20)
project(leviathan_llama_cuda LANGUAGES C CXX)

set(LLAMA_SOURCE_DIR "" CACHE PATH "Pinned llama.cpp source tree")
option(LEVIATHAN_CUDA "Build the llama.cpp CUDA backend" ON)
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
    message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to the pinned llama.cpp checkout")
endif()
set(GGML_CUDA ${LEVIATHAN_CUDA} CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
add_library(leviathan_llama_cuda SHARED backend.cpp)
target_compile_features(leviathan_llama_cuda PRIVATE cxx_std_17)
target_link_libraries(leviathan_llama_cuda PRIVATE llama)
